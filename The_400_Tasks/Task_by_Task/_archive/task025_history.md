## Rule

Line-projection from full rows/cols (lpfl): identify rows and columns where all active cells share a single color ("full" rows/cols). For each such full row, project its color upward and downward through the active region using directional cumulative-max (MaxPool). Same for full columns extending left/right. The output paints the original full rows/cols plus the projected extension cells.

## Algorithm sketch

```python
# Phase A: setup
input16 = input.astype(np.float16)          # [1,10,30,30]
colors  = input16[:, 1:, :, :]              # [1,9,30,30] ch1..9
active  = (input16.sum(axis=1, keepdims=True) > 0).astype(np.float16)  # [1,1,30,30]
row_total  = active.sum(axis=3, keepdims=True)   # [1,1,30,1] how many active cells per row
col_total  = active.sum(axis=2, keepdims=True)   # [1,1,1,30]
row_counts = colors.sum(axis=3, keepdims=True)   # [1,9,30,1] how many of each color per row
col_counts = colors.sum(axis=2, keepdims=True)   # [1,9,1,30]

# Phase B: find full rows/cols
row_active_b = (row_total > 0)              # [1,1,30,1] BOOL row has active cells
col_active_b = (col_total > 0)             # [1,1,1,30] BOOL
row_full_b   = (np.abs(row_counts - row_total) < 0.5)   # [1,9,30,1] all active cells in row = color c
row_full     = row_full_b * row_active_b   # zero out rows with no active cells
col_full_b   = (np.abs(col_counts - col_total) < 0.5)
col_full     = col_full_b * col_active_b

# Phase C: shift full-row/col masks by 1 in each direction
row_full_up   = np.pad(row_full[:,:,1:,:],  ((0,0),(0,0),(0,1),(0,0)))  # shift up (r->r-1)
row_full_down = np.pad(row_full[:,:,:-1,:], ((0,0),(0,0),(1,0),(0,0)))  # shift down
col_full_left  = np.pad(col_full[:,:,:,1:],  ((0,0),(0,0),(0,0),(0,1)))
col_full_right = np.pad(col_full[:,:,:,:-1], ((0,0),(0,0),(0,0),(1,0)))

# Phase D: directional cumulative max (MaxPool equivalent)
from scipy.ndimage import maximum_filter1d
cum_top    = maximum_filter1d(colors, 30, axis=2, origin=-14)  # max over cells above+self
cum_bottom = maximum_filter1d(colors, 30, axis=2, origin=+14)  # max over cells below+self
cum_left   = maximum_filter1d(colors, 30, axis=3, origin=-14)
cum_right  = maximum_filter1d(colors, 30, axis=3, origin=+14)
# simpler numpy: cum_top[c,r,col] = max(colors[c,0..r,col])
# actually use MaxPool(colors, kernel=(30,1), pad=(29,0,0,0)) = max of 30 cells from above

row_lines    = row_full  * active             # [1,9,30,30] full rows × active cells (broadcast)
col_lines    = col_full  * active             # [1,9,30,30]
proj_above   = cum_top    * row_full_up       # [1,9,30,30] color projected downward from above full row
proj_below   = cum_bottom * row_full_down
proj_left    = cum_left   * col_full_left
proj_right   = cum_right  * col_full_right

# Phase E: combine
nonbg_raw = row_lines + col_lines + proj_above + proj_below + proj_left + proj_right
nonbg_b   = (nonbg_raw > 0.5)               # [1,9,30,30] BOOL
nonbg_f16 = nonbg_b.astype(np.float16)
any_nonbg = nonbg_f16.max(axis=1, keepdims=True)  # [1,1,30,30]
ch0       = active - any_nonbg               # [1,1,30,30]
output    = np.concatenate([ch0, nonbg_f16], axis=1)  # [1,10,30,30]
```
